// Search customers by phone function searchCustomerByPhone() { const phoneInput = document.getElementById('genCustomerPhone').value.trim(); const autocompleteList = document.getElementById('phoneAutoComplete'); console.log('Search phone input:', phoneInput); console.log('Total customers available:', allCustomers.length); // Show list even with 1 character if (!phoneInput || phoneInput.length < 1) { autocompleteList.classList.remove('show'); return; } // Normalize input - keep only digits const normalizedInput = phoneInput.replace(/\D/g, ''); console.log('Normalized input:', normalizedInput); // Filter matching customers const matchingCustomers = allCustomers.filter(c => { if (!c.phone) { console.log('Customer has no phone:', c.full_name); return false; } // Normalize customer phone - keep only digits const normalizedPhone = c.phone.replace(/\D/g, ''); const isMatch = normalizedPhone.includes(normalizedInput); console.log('Comparing:', normalizedInput, 'with', normalizedPhone, '(', c.full_name, '):', isMatch); return isMatch; }); console.log('Matching customers found:', matchingCustomers.length); if (matchingCustomers.length === 0) { autocompleteList.innerHTML = '
No customers found with this number
'; autocompleteList.classList.add('show'); return; } // Display matching customers autocompleteList.innerHTML = matchingCustomers.map((customer, index) => `
📱 ${customer.phone}
${customer.full_name}${customer.city ? ' • ' + customer.city : ''}
`).join(''); autocompleteList.classList.add('show'); } // Select customer from autocomplete function selectCustomer(index) { const phoneInput = document.getElementById('genCustomerPhone').value.trim(); const normalizedInput = phoneInput.replace(/\D/g, ''); const matchingCustomers = allCustomers.filter(c => { if (!c.phone) return false; const normalizedPhone = c.phone.replace(/\D/g, ''); return normalizedPhone.includes(normalizedInput); }); if (matchingCustomers[index]) { const customer = matchingCustomers[index]; selectedCustomer = customer; console.log('Selected customer:', customer); document.getElementById('genCustomerPhone').value = customer.phone || ''; document.getElementById('genCustomerName').value = customer.full_name || ''; document.getElementById('genCustomerEmail').value = customer.email || ''; document.getElementById('genCustomerCity').value = customer.city || ''; document.getElementById('phoneAutoComplete').classList.remove('show'); } }
